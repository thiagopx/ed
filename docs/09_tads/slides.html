int my_strlen(const char *str)
{
    int length = 0;

    while (str[length] != '\0')
        length++;

    return length;
}

02_circle/src/point2d.c
Point2D *create_point(double x, double y) {
  Point2D *new_point = (Point2D *)malloc(sizeof(Point2D));
  if (new_point == NULL) {
    perror("Failed to allocate memory for Point2D");
    exit(EXIT_FAILURE);
  }
  new_point->x = x;
  new_point->y = y;
  return new_point;
}

O cliente usa só os headers

O cliente lê e altera o ponto só por get_point e set_point. Não toca em x, y, center ou radius.

$ make run ./test_circle Point: (2.0, 1.0) The point is inside the circle. Point: (6.0, 0.0) The point is outside the circle.
02_circle/test_circle.c
    Circle *my_circle = create_circle(0, 0, 5);

    // Create a point with coordinates (2, 1)
    Point2D *my_point = create_point(2, 1);

    // Read the coordinates back with get_point
    double x, y;
    get_point(my_point, &x, &y);
    printf("Point: (%.1f, %.1f)\n", x, y);

    // Move the point to (6, 0) with set_point and check again
    set_point(my_point, 6, 0);
    get_point(my_point, &x, &y);
    printf("Point: (%.1f, %.1f)\n", x, y);

    // Deallocate memory
    free_circle(my_circle);
    free_point(my_point);
