void ll_append(LinkedList *l, int v)
{
    ListNode *node = (ListNode *)malloc(sizeof(ListNode));
    node->info = v;
    node->next = NULL;

    if (l->first == NULL)
    {
        l->first = node;
        return;
    }

    ListNode *p = l->first;
    while (p->next != NULL)
        p = p->next;
    p->next = node;
}

Inserir em ordem: guardar o anterior

prev passa a apontar para o novo nó; prev == NULL recai no início.

11_linkedlist/01_linkedlist/linkedlist.c
void ll_insert_sorted(LinkedList *l, int v)
{
    ListNode *prev = NULL;
    ListNode *p = l->first;
    while (p != NULL && p->info < v)
    {
        prev = p;
        p = p->next;
    }
    ListNode *node = (ListNode *)malloc(sizeof(ListNode));
    node->info = v;
    if (prev == NULL)
    {
        node->next = l->first;
        l->first = node;
    }
    else
    {
        node->next = prev->next;
        prev->next = node;
    }
}

Refazer o encadeamento, depois free

ll_remove retira a primeira ocorrência e devolve 1 se achou, 0 se não.

prev == NULL→l->first = p->next
senão→prev->next = p->next
nos dois casos→só então free(p)
11_linkedlist/01_linkedlist/linkedlist.c
int ll_remove(LinkedList *l, int v)
{
    ListNode *p = l->first;
    ListNode *prev = NULL;
    while (p != NULL)
    {
        if (p->info == v)
        {
            if (prev == NULL)
                l->first = p->next;
            else
                prev->next = p->next;
            free(p);
            return 1;
        }
        else
        {
            prev = p;
            p = p->next;
        }
    }
    return 0;
}

Liberar: guardar o próximo antes do free

Depois de free(p), ler p->next seria acessar memória que não é mais do programa.

t = p->next1º: guarda o próximo
↓
free(p)2º: libera o atual
↓
p = t3º: avança
11_linkedlist/01_linkedlist/linkedlist.c
void ll_free(LinkedList *l)
{
    ListNode *p = l->first;
    while (p != NULL)
    {
        ListNode *t = p->next;
        free(p);
        p = t;
    }
    free(l);
}

Uma lista a partir de um vetor

Guardar o último nó criado evita percorrer a lista a cada valor: O(n) no total, não O(n²).

ll_append a cada valor→O(n²)
guardando o último→O(1) por valor
exercício→ll_reversed e ll_concatenate
11_linkedlist/01_linkedlist/linkedlist.c
LinkedList *ll_from_array(int v[], int n)
{
    LinkedList *l = ll_create();
    ListNode *last = NULL;
    for (int i = 0; i < n; i++)
    {
        ListNode *node = (ListNode *)malloc(sizeof(ListNode));
        node->info = v[i];
        node->next = NULL;
        if (last == NULL)
            l->first = node;
        else
            last->next = node;
        last = node;
    }
    return l;
}

void ll_remove_duplicates(LinkedList *l)
{
    for (ListNode *p = l->first; p != NULL; p = p->next)
    {
        ListNode *prev = p;
        ListNode *q = p->next;
        while (q != NULL)
        {
            if (q->info == p->info)
            {
                prev->next = q->next;
                free(q);
                q = prev->next;
            }
            else
            {
                prev = q;
                q = q->next;
            }
        }
    }
}

int ll_remove(LinkedList *l, int v)
{
    ListNode *prev = NULL;
    ListNode *p = l->first;
    while (p != NULL && p->info != v)
    {
        prev = p;
        p = p->next;
    }
    if (p == NULL)
        return 0;

    if (prev == NULL)
        l->first = p->next;
    else
        prev->next = p->next;

    if (p == l->last)
        l->last = prev;

    free(p);
    return 1;
}

void ll_join(LinkedList *l1, LinkedList *l2)
{
    if (l2->first == NULL)
        return;

    if (l1->first == NULL)
        l1->first = l2->first;
    else
        l1->last->next = l2->first;
    l1->last = l2->last;

    l2->first = l2->last = NULL;
}

Remover e devolver o novo primeiro

Quem chama reatribui o retorno — é isso que refaz o encadeamento.

p == NULL→nada a remover: devolve p
p->info == val→libera o nó e devolve p->next
senão→p->next = _ll_remove(p->next, val)
11_linkedlist/02_linkedlist-recursive/linkedlist.c
static ListNode *_ll_remove(ListNode *p, int val)
{
   ListNode *aux;
   if (p != NULL)
   {
      if (p->info == val)
      {
         aux = p;
         p = p->next;
         free(aux);
      }
      else
         p->next = _ll_remove(p->next, val);
   }
   return p;
}

void ll_remove(LinkedList *l, int val)
{
   l->first = _ll_remove(l->first, val);
}

Duas listas são iguais?

Três casos, e o terceiro delega aos subencadeamentos.

NULL e NULL→iguais
só um NULL→diferentes
senão→mesma info e restos iguais
11_linkedlist/02_linkedlist-recursive/linkedlist.c
static int _ll_equals(ListNode *p1, ListNode *p2)
{
   if (p1 == NULL && p2 == NULL) // Both ended together: equal.
      return 1;
   if (p1 == NULL || p2 == NULL) // Only one ended: different sizes.
      return 0;
   return p1->info == p2->info && _ll_equals(p1->next, p2->next);
}

int ll_equals(LinkedList *l1, LinkedList *l2)
{
   return _ll_equals(l1->first, l2->first);
}

static int _ll_count_even(ListNode *p, int *sum)
{
   if (p == NULL)
      return 0;
   int count = _ll_count_even(p->next, sum);
   if (p->info % 2 == 0)
   {
      *sum += p->info;
      count++;
   }
   return count;
}

int ll_count_even(LinkedList *l, int *sum)
{
   *sum = 0;
   return _ll_count_even(l->first, sum);
}

Inserir fecha o ciclo: O(n)

É preciso achar o último para que ele aponte para o nó novo.

11_linkedlist/03_circular-linkedlist/linkedlist.c
void ll_insert(LinkedList *l, int v)
{
    ListNode *node = (ListNode *)malloc(sizeof(ListNode));
    node->info = v;

    if (l->first == NULL)
    {
        node->next = node;
    }
    else
    {
        ListNode *last = l->first;
        while (last->next != l->first)
            last = last->next;

        node->next = l->first;
        last->next = node;
    }
    l->first = node;
}

int ll_josephus(LinkedList *l, int k)
{
    ListNode *prev = l->first;
    while (prev->next != l->first)
        prev = prev->next;

    while (prev->next != prev)
    {
        for (int i = 1; i < k; i++)
            prev = prev->next;

        ListNode *victim = prev->next;
        prev->next = victim->next;
        if (victim == l->first)
            l->first = victim->next;
        printf("%d ", victim->info);
        free(victim);
    }
    printf("\n");

    l->first = prev;
    return prev->info;
}

Retirar sem ponteiro auxiliar

Achado o nó p, o anterior e o seguinte se ligam; as extremidades mudam first ou last.

p == first→first = p->next
senão→p->prev->next = p->next
p == last→last = p->prev
senão→p->next->prev = p->prev
11_linkedlist/04_doubly-linkedlist/linkedlist.c
void ll_remove(LinkedList *l, int v)
{
    ListNode *p = _search(l, v);

    if (p == NULL)
        return;

    if (p == l->first)
        l->first = p->next;
    else
        p->prev->next = p->next;

    if (p == l->last)
        l->last = p->prev;
    else
        p->next->prev = p->prev;

    free(p);
    l->count--;
}

void ll_reverse_in_place(LinkedList *l)
{
    ListNode *p = l->first;
    while (p != NULL)
    {
        ListNode *t = p->next;
        p->next = p->prev;
        p->prev = t;
        p = t;
    }
    ListNode *t = l->first;
    l->first = l->last;
    l->last = t;
}

static ListNode *_first_by_name(LinkedList *l)
{
    if (l->first == NULL)
        return NULL;
    ListNode *best = l->first;
    for (ListNode *p = l->first->next; p != NULL; p = p->next)
        if (strcmp(st_name(p->info), st_name(best->info)) < 0)
            best = p;
    return best;
}

static ListNode *_at(LinkedList *l, int pos)
{
    if (pos < 0 || pos >= l->count)
        return NULL;
    ListNode *p = l->first;
    for (int i = 0; i < pos; i++)
        p = p->next;
    return p;
}
