struct stack
{
    int n_max; // Capacity of v.
    int n;     // Number of elements; the top is v[n-1].

Stack *s_create()
{
    Stack *s = (Stack *)malloc(sizeof(Stack));
    s->n = 0;
    s->n_max = 4;
    s->v = (float *)malloc(s->n_max * sizeof(float));
    return s;
}

Pilha com vetor

Empilhar: v[n++], e realocar se cheio

O(1) quando há espaço; O(n) na vez que realoca, porque o realloc pode copiar tudo.

n < n_max→grava em v[n]: O(1)
n == n_max→reallocate dobra o vetor: O(n)
v[s->n++]→usa o n antigo como índice, depois incrementa
12_stack-and-queues/01_stack-array/stack.c
static void reallocate(Stack *s)
{
    s->n_max *= 2;
    s->v = (float *)realloc(s->v, s->n_max * sizeof(float));
}

void s_push(Stack *s, float v)
{
    if (s->n == s->n_max)
        reallocate(s);

    s->v[s->n++] = v;
}

Desempilhar: v[--n]

Primeiro decrementa, depois lê. Nada é apagado: o valor fica fora da parte ocupada.

s_pop→v[--n]: O(1)
s_top→v[n-1], sem mudar n: O(1)
s_is_empty→n == 0: O(1)

float s_top(Stack *s)
{
    assert(!s_is_empty(s));
    return s->v[s->n - 1];
}

float s_pop(Stack *s)
{
    assert(!s_is_empty(s));
    return s->v[--s->n];
}

Pilha com vetor

Ficha: pilha com vetor

Vantagens e desvantagens desta implementação.

vantagens

  • push, pop e top custam O(1): o topo está sempre em v[n-1], com acesso direto.
  • Os elementos ficam contíguos, e o percurso aproveita a cache.
  • São só duas alocações no total, a estrutura e o vetor, em vez de uma por elemento.

desvantagens

  • O push que encontra o vetor cheio custa O(n), porque o realloc pode copiar tudo.
  • Logo depois de dobrar, até metade do vetor fica reservada sem uso.
  • O vetor nunca encolhe: depois de um pico, a memória fica reservada até s_free.

Pilha com vetor

Ficha: pilha com vetor

O que a estrutura não permite, e o que a linguagem C impõe.

limitações da estrutura

  • Só o topo é acessível: para ver o terceiro, é preciso desempilhar os dois de cima.
  • Retirar da pilha vazia aborta: o cliente testa antes com s_is_empty.

limitações de C

  • A pilha só guarda float; outro tipo exige outro TAD ou void *.
  • O realloc pode mover o vetor: um ponteiro antigo para v[i] fica pendente.
  • Não há coleta de lixo: quem cria a pilha precisa chamar s_free.

E se o push nunca custasse O(n)?

A pilha com lista troca o vetor por nós: cada push aloca só um nó, e nada é copiado.

com vetor→o push com o vetor cheio realoca e copia tudo: O(n)
com lista→todo push aloca um nó e liga ao topo: O(1)

float s_top(Stack *s)
{
    assert(!s_is_empty(s));
    return s->top->info;
}

float s_pop(Stack *s)
{
    assert(!s_is_empty(s));

    float v = s->top->info;
    StackNode *p = s->top; // Saved to be freed after top moves on.
    s->top = s->top->next;
    free(p);
    return v;
}

Pilha com lista

Liberar: um free por nó

Guarda o próximo antes de cada free, como o ll_free do módulo 11: O(n).

t = p->next→guarda o próximo
free(p)→libera o atual
free(s)→por fim, a estrutura
12_stack-and-queues/02_stack/stack.c
void s_free(Stack *s)
{
    StackNode *p = s->top;
    while (p != NULL)
    {
        StackNode *t = p->next;
        free(p);
        p = t;
    }
    free(s);
}

Pilha com lista

Ficha: pilha com lista

Vantagens e desvantagens desta implementação.

vantagens

  • push, pop e top custam O(1) sempre: não existe a inserção cara da realocação.
  • A memória acompanha n: a pilha cresce e encolhe um nó por vez.
  • Nenhum elemento muda de endereço enquanto está na pilha.

desvantagens

  • Cada float de 4 bytes carrega um next de 8 bytes, e cada push faz uma alocação.
  • Liberar a pilha custa O(n), porque cada nó precisa do seu próprio free.
  • Os nós ficam espalhados pela memória, e o percurso fica mais lento que no vetor.

{

Problema: cancelar vizinhos iguais

Só o topo pode cancelar o valor novo: 5 3 3 5 7 vira 7; 1 2 2 1 some; 1 2 1 2 não muda.

pilha vazia→o && nem chama s_top
topo == v[i]→s_pop: os dois somem
senão→s_push(v[i])
no fim→o que sobra na pilha é a resposta
código cliente: cancel.c
int cancel_pairs(float v[], int n)
{
    Stack *s = s_create();
    for (int i = 0; i < n; i++)
    {
        if (!s_is_empty(s) && s_top(s) == v[i])
            s_pop(s);                           // v[i] cancels the top
        else
            s_push(s, v[i]); // nothing to cancel: v[i] waits on the top
    }
    int left = 0;
    while (!s_is_empty(s))
    {
        printf("%.0f ", s_pop(s));
        left++;
    }
    printf("(%d left)\n", left);
    s_free(s);
    return left;
}

struct calc
{

    Stack *s;     // Stack of operands (any Stack implementation works)
};

Calc *calc_create(const char *fmt)
{
    Calc *c = (Calc *)malloc(sizeof(Calc));
    strncpy(c->fmt, fmt, 20);
    c->fmt[20] = '\0';
    c->s = s_create();
    return c;
}

Se faltar operando, ele vale zero: o teste evita o assert de s_pop.

v2→o topo, desempilhado primeiro
v1→o de baixo
pilha vazia→0.0f, sem chamar s_pop

void calc_operator(Calc *c, char op)
{
    float v2 = s_is_empty(c->s) ? 0.0f : s_pop(c->s);
    float v1 = s_is_empty(c->s) ? 0.0f : s_pop(c->s);
    float v = 0.0f;

    switch (op)
    {
    case '+': v = v1 + v2; break;
    case '-': v = v1 - v2; break;
    case '*': v = v1 * v2; break;
    case '/': v = v1 / v2; break;
    }

    s_push(c->s, v);
    printf(c->fmt, v);
}

head, n e o tail calculado

head é o próximo a sair; tail = (head + n) % n_max é onde entra o próximo.

head→índice do primeiro (CLRS)
tail→não é guardado: (head + n) % n_max

void q_enqueue(Queue *q, float v)
{
    if (q->n == q->n_max)
    {
        reallocate(q);

        // If the queue wraps around the end of the old vector, the block from head to
        // the old end must move to the end of the new vector. memmove (string.h) copies
        // a block of bytes, like a loop copying element by element.
        if (q->head != 0)
        {
            memmove(
                &q->v[q->n_max - q->n + q->head], // destination
                &q->v[q->head],                   // source
                (q->n - q->head) * sizeof(float)  // bytes to shift
            );
            q->head = q->n_max - q->n + q->head;
        }
    }
    int tail = (q->head + q->n) % q->n_max;
    q->v[tail] = v;
    q->n++;
}

float q_dequeue(Queue *q)
{
   assert(!q_is_empty(q));

   float v = q->head->info;
   QueueNode *p = q->head; // Saved to be freed after head moves on.

   if (q->head != q->tail)
      q->head = q->head->next;
   else

   return v;
}

Problema: girar a fila dupla

Retirar de uma ponta e inserir na outra: 1 2 3 4 5 com k = 2 dá 3 4 5 1 2; depois, k = -3 dá 5 1 2 3 4.

1 2 3 4 5início
↓

↓
5 1 2 3 4k = -3: pop_back → push_front
código cliente: rotate.c
void rotate(Deque *d, int k)
{
    if (dq_is_empty(d))
        return;
    for (; k > 0; k--)
        dq_push_back(d, dq_pop_front(d));
    for (; k < 0; k++)
        dq_push_front(d, dq_pop_back(d));
}

Para reconhecer a estrutura

Pergunte quem precisa sair primeiro e simule com os dados, indicando as pontas a cada passo.

sai o mais recente: pilha

  • Voltar pelo caminho num labirinto: o último passo dado é o primeiro a desfazer.
  • Retomar a função que chamou a atual, como faz a pilha de execução.
  • Casar cada item com o anterior pendente, como ao cancelar vizinhos.

sai o mais antigo, ou pelas pontas

  • Atender na ordem de chegada pede uma fila: quem chegou primeiro sai primeiro.
  • Manter a janela das últimas medidas pede uma fila: sai a mais antiga.
  • Girar, ou consumir pelas duas extremidades, pede uma fila dupla.
